    printf("Tarifa por hora: S/ %.2f\n", tarifa);

\textbf{Procedimiento:} Se almacenan diez mediciones en un arreglo y se calcula su suma. La media se obtiene dividiendo la suma entre diez. Después se cuentan los valores mayores que la media.
\begin{lstlisting}[style=codigo,language=C++]
#include <iostream>
using namespace std;

int main() {
    double datos[10], suma = 0, media;
    int contador = 0;

    for (int i = 0; i < 10; i++) {
        cout << "Ingrese la medicion " << i + 1 << ": ";
        cin >> datos[i];

        suma += datos[i];
    }

    media = suma / 10;

    for (int i = 0; i < 10; i++) {
        if (datos[i] > media) {
            contador++;
        }
    }

    cout << "Media aritmetica: " << media << endl;
    cout << "Observaciones por encima de la media: "
         << contador << endl;

    return 0;
}

\section*{Ejercicio 11. Validación de contraseña (C++)}
\textbf{Procedimiento:} El ciclo while solicita la clave hasta que sea correcta. Se incrementa el contador cada vez que se realiza un intento. Para este ejemplo, la clave correcta se establece en 1234.
\begin{lstlisting}[style=codigo,language=C++]
#include <iostream>
using namespace std;

int main() {
    const int CLAVE_CORRECTA = 1234;
    int clave, intentos = 0;

    cout << "Ingrese la clave numerica: ";
    cin >> clave;
    intentos++;

    while (clave != CLAVE_CORRECTA) {
        cout << "Clave incorrecta. Intente nuevamente.\n";

        cout << "Ingrese la clave numerica: ";
        cin >> clave;

        intentos++;
    }

    cout << "Acceso permitido.\n";
    cout << "Numero de intentos: " << intentos << endl;

    return 0;
}

\section*{Ejercicio 12. Procesamiento de mediciones con break y continue (C++)}
\textbf{Procedimiento:} Se leen como máximo diez valores. Los negativos se omiten con continue; el valor 999 detiene el ciclo con break. Solo se cuentan las mediciones no negativas distintas de 999.
\begin{lstlisting}[style=codigo,language=C++]
#include <iostream>
using namespace std;

int main() {
    double valor;
    int validos = 0;

    for (int i = 0; i < 10; i++) {
        cout << "Ingrese la medicion " << i + 1
             << " (999 para terminar): ";
        cin >> valor;

        if (valor == 999) {
            break;
        }

        if (valor < 0) {
            cout << "Valor negativo: se omite.\n";
            continue;
        }

        validos++;
    }

    cout << "Valores validos procesados: "
         << validos << endl;

    return 0;
}

\section*{Ejercicio 13. Análisis de edades mediante un arreglo}
\textbf{Procedimiento:} Se almacenan diez edades. Durante el ingreso se determina la suma, la edad mínima y la máxima. Después se calcula la media y se cuentan las edades superiores a ella.
\begin{lstlisting}[style=codigo,language=C]
#include <stdio.h>

int main(void) {
    int edades[10], suma = 0;
    int minima, maxima, superiores = 0;
    float media;

    for (int i = 0; i < 10; i++) {
        printf("Edad del participante %d: ", i + 1);
        scanf("%d", &edades[i]);

        if (i == 0) {
            minima = maxima = edades[i];
        } else {
            if (edades[i] < minima) {
                minima = edades[i];
            }

            if (edades[i] > maxima) {
                maxima = edades[i];
            }
        }

        suma += edades[i];
    }

    media = (float)suma / 10;

    for (int i = 0; i < 10; i++) {
        if (edades[i] > media) {
            superiores++;
        }
    }

    printf("\nEdad minima: %d\n", minima);
    printf("Edad maxima: %d\n", maxima);
    printf("Media de edades: %.2f\n", media);
    printf("Participantes por encima de la media: %d\n",
           superiores);

    return 0;
}
\end{lstlisting}

\section*{Ejercicio 14. Comparación de complejidad O(n) y O(n²)}
\textbf{Procedimiento:} El primer procedimiento ejecuta una operación por cada elemento, por lo que realiza n operaciones. El segundo ejecuta dos ciclos anidados de n iteraciones cada uno, por lo que realiza n por n, es decir, n al cuadrado operaciones.
\begin{lstlisting}[style=codigo,language=C]
#include <stdio.h>

int main(void) {
    int n;
    long long operacionesLineales = 0;
    long long operacionesCuadraticas = 0;

    printf("Ingrese el numero de elementos: ");
    scanf("%d", &n);

    if (n <= 0) {
        printf("n debe ser un entero positivo.\n");
        return 1;
    }

    /* Procedimiento 1: un recorrido */
    for (int i = 0; i < n; i++) {
        operacionesLineales++;
    }

    /* Procedimiento 2: dos ciclos anidados */
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            operacionesCuadraticas++;
        }
    }

    printf("Operaciones del primer procedimiento: %lld\n",
           operacionesLineales);
    printf("Complejidad aproximada: O(n)\n");

    printf("Operaciones del segundo procedimiento: %lld\n",
           operacionesCuadraticas);
    printf("Complejidad aproximada: O(n^2)\n");

    return 0;
}
\end{lstlisting}

\section*{Ejercicio 15. Funciones para promedio y conteo}
\textbf{Procedimiento:} La función calcular\_promedio recibe tres mediciones y retorna su media. La función contar\_superiores recibe las mediciones y el promedio, y retorna cuántas superan dicho promedio.
\begin{lstlisting}[style=codigo,language=C]
#include <stdio.h>

float calcular_promedio(float a, float b, float c) {
    return (a + b + c) / 3.0f;
}

int contar_superiores(float a, float b, float c,
                      float promedio) {
    int contador = 0;

    if (a > promedio) contador++;
    if (b > promedio) contador++;
    if (c > promedio) contador++;

    return contador;
}

int main(void) {
    float m1, m2, m3, promedio;
    int cantidad;

    printf("Ingrese la primera medicion: ");
    scanf("%f", &m1);

    printf("Ingrese la segunda medicion: ");
    scanf("%f", &m2);

    printf("Ingrese la tercera medicion: ");
    scanf("%f", &m3);

    promedio = calcular_promedio(m1, m2, m3);

    cantidad = contar_superiores(m1, m2, m3, promedio);

    printf("Promedio: %.2f\n", promedio);
    printf("Mediciones por encima del promedio: %d\n",
           cantidad);

    return 0;
}
\end{lstlisting}
